لسانLisan

لغة برمجة عربية فصيحة مكتوبة بلغة Rust. اكتب بكلمات عربية وأرقام عربية وغربية وفارسية، وشغّلها بمفسّر أو آلة افتراضية، أو ترجمها إلى كود آلة أصلي عبر LLVM.An Arabic programming language written in Rust. Write with Arabic keywords and identifiers, use Western, Arabic-Indic or Persian digits, run it on an interpreter or VM, or compile natively through LLVM.

أصناف وكائناتClasses and objects

examples/oop_car.lsn
// مثال شامل للبرمجة كائنية التوجه (OOP) في لغة لسان

صنف سيارة {
    دالة بناء(النوع، السرعة_القصوى) {
        هذا.النوع = النوع
        هذا.السرعة_القصوى = السرعة_القصوى
        هذا.السرعة_الحالية = 0
    }

    دالة تسارع(مقدار) {
        هذا.السرعة_الحالية = هذا.السرعة_الحالية + مقدار
        إذا (هذا.السرعة_الحالية > هذا.السرعة_القصوى) {
            هذا.السرعة_الحالية = هذا.السرعة_القصوى
        }
        أرجع هذا.السرعة_الحالية
    }

    دالة فرملة(مقدار) {
        هذا.السرعة_الحالية = هذا.السرعة_الحالية - مقدار
        إذا (هذا.السرعة_الحالية < 0) {
            هذا.السرعة_الحالية = 0
        }
        أرجع هذا.السرعة_الحالية
    }

    دالة تقرير() {
        أرجع "سيارة " + هذا.النوع + " تسير بسرعة: " + نص(هذا.السرعة_الحالية) + " كم/س"
    }
}

دع س = جديد سيارة("تسلا", 250)
اطبع("تم إنشاء الكائن من صنف: " + نوع(س))
اطبع(س.تقرير())

اطبع("--- تسارع بمقدار 80 ---")
س.تسارع(80)
اطبع(س.تقرير())

اطبع("--- تسارع بمقدار 200 (يتجاوز القصوى) ---")
س.تسارع(200)
اطبع(س.تقرير())

اطبع("--- فرملة بمقدار 50 ---")
س.فرملة(50)
اطبع(س.تقرير())

اطبع("--- تصدير بيانات الكائن إلى JSON ---")
اطبع(ترميز_json(س))
